    <script>
        // Overlay Management
        const overlayBg = document.getElementById('overlay-bg');

        function showOverlay() {
            overlayBg.classList.remove('hidden');
        }

        function hideOverlay() {
            overlayBg.classList.add('hidden');
        }

        // Modal Functions
        function openModal(modalId) {
            document.getElementById(modalId).classList.remove('hidden');
            showOverlay();
        }

        function closeModal(modalId) {
            document.getElementById(modalId).classList.add('hidden');
            hideOverlay();
        }

        // Slide-over Panel Functions
        const guestPanel = document.getElementById('guest-panel');

        function openSlideOver() {
            guestPanel.classList.add('open');
            showOverlay();
        }

        function closeSlideOver() {
            guestPanel.classList.remove('open');
            hideOverlay();
        }

        // Skeleton Loader Toggle
        const skeletonLoader = document.getElementById('skeleton-loader');
        const tableContent = document.getElementById('table-content');
        let isLoading = false;

        function toggleSkeleton() {
            isLoading = !isLoading;
            if (isLoading) {
                skeletonLoader.classList.remove('hidden');
                tableContent.classList.add('hidden');
            } else {
                skeletonLoader.classList.add('hidden');
                tableContent.classList.remove('hidden');
            }
        }

        // Close modals/panels when clicking on the dark background
        overlayBg.addEventListener('click', () => {
            closeModal('export-modal');
            closeSlideOver();
        });

        // Close with Escape key
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                closeModal('export-modal');
                closeSlideOver();
            }
        });
    </script>
